# Import TensorFlow and other required libraries
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
print(tf.__version__)  # Print TensorFlow version being used

# Load the Fashion MNIST dataset
# This dataset contains 70,000 grayscale images of 10 fashion categories
# The images are 28x28 pixels, with 60,000 for training and 10,000 for testing
fashion_mnist = tf.keras.datasets.fashion_mnist
(train_images, train_labels), (test_images, test_labels) = fashion_mnist.load_data()

# Define class names for the 10 fashion categories
class_names = ['T-shirt/top', 'Trouser', 'Pullover', 'Dress', 'Coat',
               'Sandal', 'Shirt', 'Sneaker', 'Bag', 'Ankle boot']

# Examine the shape of the training data (number of images, height, width)
train_images.shape

# Check number of training labels (should match number of training images)
len(train_labels)

# View the first few training labels (each is a number 0-9 representing class)
train_labels

# Examine the shape of the test data
test_images.shape

# Check number of test labels
len(test_labels)

# Visualize the first training image with colorbar showing pixel values
plt.figure()
plt.imshow(train_images[0])
plt.colorbar()  # Shows pixel intensity scale (0-255)
plt.grid(False)  # Turn off grid lines
plt.show()

# Normalize pixel values from 0-255 to 0-1 to help neural network training
train_images = train_images / 255.0
test_images = test_images / 255.0

# Display the first 25 images from the training set with their class labels
plt.figure(figsize=(10,10))
for i in range(25):
    plt.subplot(5,5,i+1)  # Create a 5x5 grid of plots
    plt.xticks([])  # Remove x-axis ticks
    plt.yticks([])  # Remove y-axis ticks
    plt.grid(False)  # Remove grid lines
    plt.imshow(train_images[i], cmap=plt.cm.binary)  # Show image in grayscale
    plt.xlabel(class_names[train_labels[i]])  # Label with class name
plt.show()

# Build the neural network model
model = tf.keras.Sequential([
    # Flatten the 28x28 image into a 784-pixel vector
    tf.keras.layers.Flatten(input_shape=(28, 28)),
    # First dense layer with 128 neurons and ReLU activation
    tf.keras.layers.Dense(128, activation='relu'),
    # Output layer with 10 neurons (one for each class)
    tf.keras.layers.Dense(10)
])

# Compile the model with:
# - Adam optimizer (adaptive learning rate)
# - Sparse Categorical Crossentropy loss (for multi-class classification)
# - Accuracy as the evaluation metric
model.compile(optimizer='adam',
              loss=tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True),
              metrics=['accuracy'])

# Evaluate the model on the test set before training (random weights)
test_loss, test_acc = model.evaluate(test_images, test_labels, verbose=2)
print('\nTest accuracy:', test_acc)

# Create a probability model by adding a Softmax layer to convert logits to probabilities
probability_model = tf.keras.Sequential([model, 
                                       tf.keras.layers.Softmax()])

# Make predictions on the entire test set
predictions = probability_model.predict(test_images)

# View the prediction probabilities for the first test image
predictions[0]

# Get the predicted class (index with highest probability)
np.argmax(predictions[0])

# Check the true label for the first test image
test_labels[0]

# Define helper functions for visualization
def plot_image(i, predictions_array, true_label, img):
    """Plot an image with prediction and true label."""
    true_label, img = true_label[i], img[i]
    plt.grid(False)
    plt.xticks([])
    plt.yticks([])
    plt.imshow(img, cmap=plt.cm.binary)

    predicted_label = np.argmax(predictions_array)
    if predicted_label == true_label:
        color = 'blue'  # Correct prediction
    else:
        color = 'red'   # Incorrect prediction

    plt.xlabel(
        "{} {:2.0f}% ({})".format(
            class_names[predicted_label],
            100 * np.max(predictions_array),
            class_names[true_label]
        ),
        color=color
    )

def plot_value_array(i, predictions_array, true_label):
    """Plot the prediction probabilities as a bar chart."""
    true_label = true_label[i]
    plt.grid(False)
    plt.xticks(range(10))  # One tick per class
    plt.yticks([])
    thisplot = plt.bar(range(10), predictions_array, color="#777777")
    plt.ylim([0, 1])  # Probability range

    predicted_label = np.argmax(predictions_array)
    thisplot[predicted_label].set_color('red')  # Highlight predicted class
    thisplot[true_label].set_color('blue')      # Highlight true class
